<script>
  const $ = id => document.getElementById(id);
  let notesType = 'both';
  let notesVerified = 'Yes';
  let notesArchived = 'Yes';

  function getAdditionalDomains(){
    const raw = $('g_additionalDomains').value.trim();
    if(!raw) return [];
    return raw.split(/[\n,]+/).map(s=>s.trim()).filter(Boolean);
  }

  function primaryDomain(){ return $('g_domain').value.trim() || '[domain]'; }
  function domainArray(){ return [primaryDomain(), ...getAdditionalDomains()]; }

  function generate(){
    const rowId = $('notes_rowId').value.trim() || '[Row ID]';
    const typeLabel = notesType === 'website' ? 'Website' : notesType === 'email' ? 'Email' : 'Website and emails';
    const source = $('notes_source').value.trim() || '[source server]';
    const dest = $('notes_dest').value.trim() || '[destination server]';
    const diskSource = $('notes_diskSource').value.trim() || '[paste du -sch output]';
    const diskDest = $('notes_diskDest').value.trim() || '[paste du -sch output]';
    const dnsStatus = $('notes_dnsStatus').value.trim() || 'DNS updated';
    const domains = domainArray();

    const lines = [];
    lines.push('Pre-migration notes/pulse:\n');
    lines.push(`Row ID: ${rowId}\n`);
    lines.push(`What is being migrated: ${typeLabel}\n`);
    lines.push('Domains:\n');
    domains.forEach(d => lines.push(d));
    lines.push(`\nDisk usage on Source\n\n${diskSource}\n`);
    lines.push(`Source: ${source}\n`);
    lines.push(`Destination: ${dest}\n`);
    lines.push('---------------------------------------------------------------\n');
    lines.push('Post migration notes/pulse:\n');
    lines.push(`Disk usage on Destination:\n\n${diskDest}\n`);
    lines.push(`DNS Status : ${dnsStatus}\n`);
    lines.push(`Verified count of email accounts & emails listed in webmail: ${notesVerified}\n`);
    lines.push(`Source archived and old backup deleted: ${notesArchived}`);

    const text = lines.join('\n');
    $('output').textContent = text;
    $('wordCount').textContent = `${text.trim().split(/\s+/).filter(Boolean).length} words`;
    $('charCount').textContent = `${text.length} characters`;
  }

  function applyTheme(theme){
    document.documentElement.setAttribute('data-theme', theme);
    document.querySelectorAll('#themeToggle .opt').forEach(o=>o.classList.toggle('active', o.dataset.themeValue === theme));
    try{ localStorage.setItem('migrationToolTheme', theme); }catch(e){}
  }
  document.querySelectorAll('#themeToggle .opt').forEach(opt=>opt.addEventListener('click', ()=> applyTheme(opt.dataset.themeValue)));
  applyTheme(document.documentElement.getAttribute('data-theme') || 'light');

  document.querySelectorAll('#notes_type button').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      document.querySelectorAll('#notes_type button').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      notesType = btn.dataset.value;
      generate();
    });
  });
  document.querySelectorAll('#notes_verified button').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      document.querySelectorAll('#notes_verified button').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      notesVerified = btn.dataset.value;
      generate();
    });
  });
  document.querySelectorAll('#notes_archived button').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      document.querySelectorAll('#notes_archived button').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      notesArchived = btn.dataset.value;
      generate();
    });
  });

  document.querySelectorAll('input[type=text], textarea').forEach(el=>el.addEventListener('input', generate));
  $('copyBtn').addEventListener('click', async ()=>{
    try{ await navigator.clipboard.writeText($('output').textContent); }catch(e){}
    const btn = $('copyBtn');
    btn.textContent = 'Copied';
    btn.classList.add('copied');
    setTimeout(()=>{ btn.textContent = 'Copy'; btn.classList.remove('copied'); }, 1600);
  });
  $('resetBtn').addEventListener('click', ()=>{
    document.querySelectorAll('input[type=text], textarea').forEach(el=>el.value='');
    $('notes_dnsStatus').value = 'DNS updated';
    generate();
  });
  generate();
</script>
