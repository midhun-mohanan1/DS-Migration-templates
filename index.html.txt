<script>
  const $ = id => document.getElementById(id);
  let activeBrand = 'crazydomains';
  let migTypePremig = 'email';
  let premigDnsManaged = 'us';

  const TYPE_META = {
    email: { label:'email', includesEmail:true, includesWebsite:false },
    website: { label:'website', includesEmail:false, includesWebsite:true },
    both: { label:'website and email', includesEmail:true, includesWebsite:true }
  };

  const BRANDS = {
    crazydomains: {
      titanWebmail: 'https://crazydomains.titan.email/login/',
      titanDnsRecordsArticle: 'https://www.crazydomains.com.au/help/article/set-up-mx-and-txt-records-for-titan-email'
    },
    vodien: {
      titanWebmail: 'https://vodien.titan.email/login/',
      titanDnsRecordsArticle: 'https://www.vodien.com/help/article/set-up-mx-and-txt-records-for-vodien-titan-email'
    }
  };

  function formatDate(iso){
    if(!iso) return '[date]';
    const d = new Date(iso + 'T00:00:00');
    const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
    return `${d.getDate()} ${months[d.getMonth()]} ${d.getFullYear()}`;
  }

  function autoPassword(explicit, ticketId){
    const v = explicit.trim();
    if(v) return v;
    if(!$('g_autoPassword').checked) return '[password]';
    return `Migrate@${ticketId.trim() || '[Ticket ID]'}`;
  }

  function getAdditionalDomains(){
    const raw = $('g_additionalDomains').value.trim();
    if(!raw) return [];
    return raw.split(/[\n,]+/).map(s=>s.trim()).filter(Boolean);
  }

  function primaryDomain(){
    return $('g_domain').value.trim() || '[domain]';
  }

  function domainArray(){
    return [primaryDomain(), ...getAdditionalDomains()];
  }

  function domainList(){
    const list = domainArray();
    if(list.length === 1) return list[0];
    if(list.length === 2) return `${list[0]} and ${list[1]}`;
    return `${list.slice(0,-1).join(', ')} and ${list[list.length-1]}`;
  }

  function countsFindingParagraph(){
    const emailCount = $('premig_emailCount').value.trim();
    const websiteCount = $('premig_websiteCount').value.trim();
    const emailList = $('premig_emailList').value.trim();
    const domain = primaryDomain();
    const lines = [];

    if(emailList){
      const addresses = emailList.split('\n').map(s=>s.trim()).filter(Boolean);
      const one = addresses.length === 1;
      lines.push(`There ${one ? 'is' : 'are'} currently ${addresses.length} email account${one ? '' : 's'} hosted on the source plan:\n\n${addresses.join('\n')}`);
    } else if(emailCount){
      const num = parseFloat(emailCount);
      if(!isNaN(num) && num === 0){
        lines.push('There are currently no email accounts hosted on the source plan.');
      } else {
        const one = !isNaN(num) && num === 1;
        lines.push(`There ${one ? 'is' : 'are'} currently ${emailCount} email account${one ? '' : 's'} hosted on the source plan under the domain ${domain}.`);
      }
    }

    if(websiteCount){
      const num = parseFloat(websiteCount);
      const one = !isNaN(num) && num === 1;
      lines.push(`There ${one ? 'is' : 'are'} ${websiteCount} website${one ? '' : 's'} hosted on the source plan.`);
    }

    lines.push(premigDnsManaged === 'us'
      ? 'The DNS/NS records are currently managed by us.'
      : "The domain's Name Server (NS) records are currently pointing to name servers that are not managed by us.");

    const hasCounts = emailCount || emailList || websiteCount;
    return hasCounts
      ? `We have completed the pre-migration checks and can confirm the following:\n\n${lines.join('\n')}`
      : lines.join('\n');
  }

  function findingParagraphPremig(){
    if($('premig_customFindingToggle').checked){
      return $('premig_customFinding').value.trim() || null;
    }
    if(!$('premig_includeCounts').checked){
      return premigDnsManaged === 'us'
        ? 'The DNS/NS records are currently managed by us.'
        : "As part of our pre-migration checks, we found that the domain's Name Server (NS) records are currently pointing to name servers that are not managed by us.";
    }
    return countsFindingParagraph();
  }

  function dnsInstructionsPremig(){
    const meta = TYPE_META[migTypePremig];
    const brand = BRANDS[activeBrand];
    const p = [];

    if(premigDnsManaged === 'us'){
      const byType = {
        email: "Once the email migration has been completed, we will need to update the DNS records to point the domain's email services to the new mail server.",
        website: 'Once the website migration has been completed, we will need to update the relevant DNS records so that the website loads from the new server.',
        both: 'Once the website and email migration has been completed, we will need to update the DNS records to point the domain to the new hosting plan.'
      };
      p.push(byType[migTypePremig]);
      p.push('Please note that this DNS change will require propagation, which can take approximately 4\u20136 hours to complete worldwide.');
      if(meta.includesEmail){
        p.push('If you are using an email client such as Outlook, Thunderbird, Apple Mail, or a mobile mail application, you may need to reconfigure the email account using the new mail server settings after the DNS records have been updated and propagation has completed.');
      }
    } else {
      p.push("Since the domain's Name Servers are managed externally, we will not be able to make the required DNS changes on your behalf.");
      if($('premig_recordsNow').checked){
        const records = $('premig_records').value.trim() || '[DNS records]';
        p.push(`Please update the following DNS records with your current DNS provider:\n\n${records}`);
        if($('premig_titanEnable').checked){
          p.push(`You can also find the required DNS records and setup instructions in the General Instructions section of the following article:\n${brand.titanDnsRecordsArticle}`);
          p.push('Please ensure that any existing MX records are removed or replaced with the above records to avoid conflicts.');
        }
      } else {
        p.push('Once the migration is completed, we will provide you with the required DNS records and their corresponding values, which you can then update with your current DNS provider.');
      }
      p.push('We will keep you updated once the migration is complete and the DNS records are ready to be updated.');
    }
    return p;
  }

  function generate(){
    const meta = TYPE_META[migTypePremig];
    const currentPlan = $('premig_currentPlan').value.trim();
    const destPlan = $('premig_destPlan').value.trim();
    const brand = BRANDS[activeBrand];
    const scheduled = $('premig_scheduled').checked;

    const p = [];
    if($('premig_thankYou').checked) p.push('Thank you for your patience.');

    let opening = `We have received your ${meta.label} migration request for ${domainList()}`;
    if(currentPlan && destPlan) opening += `, from the ${currentPlan} to the ${destPlan}`;
    opening += '.';
    p.push(opening);

    const finding = findingParagraphPremig();
    if(finding) p.push(finding);

    const nsChecked = $('premig_nsChecker').checked;
    const mxChecked = $('premig_mxChecker').checked;
    if(nsChecked || mxChecked){
      const domain = primaryDomain();
      const linkLines = [`You can view the current ${nsChecked && mxChecked ? 'NS and MX records' : nsChecked ? 'NS records' : 'MX records'} using the following DNS Checker link${nsChecked && mxChecked ? 's' : ''}:`];
      if(nsChecked) linkLines.push(`https://dnschecker.org/#NS/${domain}`);
      if(mxChecked) linkLines.push(`https://dnschecker.org/#MX/${domain}`);
      p.push(linkLines.join('\n'));
    }

    if($('premig_titanEnable').checked){
      const multiple = $('premig_titanMultiple').checked;
      const password = autoPassword($('premig_titanPassword').value, $('g_ticketId').value);
      const emailLine = multiple ? 'Your respective email address' : ($('premig_titanEmail').value.trim() || '[mailbox email]');
      p.push('We have created the email account(s) on the TITAN dashboard.');
      if(multiple){
        p.push(`You can access the new mailboxes using the Titan Webmail URL below:\n\nWebmail: ${brand.titanWebmail}\nEmail: ${emailLine}\nTemporary Password: ${password}`);
        p.push(`As the passwords for the existing email accounts were not provided, we will reset the passwords temporarily to proceed with the migration.\n\nThe temporary password for the migration is:\n\n${password}`);
      } else {
        p.push(`You can access the new mailbox using the Titan Webmail URL below:\n\nWebmail: ${brand.titanWebmail}\nEmail: ${emailLine}\nPassword: ${password}`);
      }
    }

    dnsInstructionsPremig().forEach(line => p.push(line));

    if($('premig_screenshots').checked){
      const byType = {
        email: 'We have also attached screenshots showing the current email account status and DNS configuration for your reference.',
        website: 'We have also attached screenshots showing the current website status and DNS configuration for your reference.',
        both: 'We have also attached screenshots showing the current website, email account status, and DNS configuration for your reference.'
      };
      p.push(byType[migTypePremig]);
    }

    if(scheduled){
      const time = $('premig_time').value.trim() || '[time]';
      const tz = $('premig_timezone').value.trim() || '[TZ]';
      p.push(`The migration is currently scheduled for:\nDate: ${formatDate($('premig_date').value)}\nTime: After ${time} (${tz})`);
      p.push('We will proceed with the migration as scheduled unless we hear otherwise from you.');
    }

    if($('premig_contactClose').checked){
      p.push("If you have any questions or require any assistance with the DNS update, please feel free to contact us. We'll be happy to assist.");
    }

    if($('premig_includeSignoff').checked){
      const signoff = $('premig_signoffText').value.trim();
      if(signoff) p.push(signoff);
    }

    const text = p.join('\n\n');
    $('output').textContent = text;
    const words = text.trim().split(/\s+/).filter(Boolean).length;
    $('wordCount').textContent = `${words} words`;
    $('charCount').textContent = `${text.length} characters`;
  }

  // Theme toggle
  function applyTheme(theme){
    document.documentElement.setAttribute('data-theme', theme);
    document.querySelectorAll('#themeToggle .opt').forEach(o=>{
      o.classList.toggle('active', o.dataset.themeValue === theme);
    });
    try{ localStorage.setItem('migrationToolTheme', theme); }catch(e){}
  }
  document.querySelectorAll('#themeToggle .opt').forEach(opt=>{
    opt.addEventListener('click', ()=> applyTheme(opt.dataset.themeValue));
  });
  applyTheme(document.documentElement.getAttribute('data-theme') || 'light');

  // Brand toggle
  document.querySelectorAll('#brandToggle button').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      document.querySelectorAll('#brandToggle button').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      activeBrand = btn.dataset.value;
      generate();
    });
  });

  // Migration type toggle
  document.querySelectorAll('#migType_premig button').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      document.querySelectorAll('#migType_premig button').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      migTypePremig = btn.dataset.value;
      generate();
    });
  });

  document.querySelectorAll('#premig_dnsManaged button').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      document.querySelectorAll('#premig_dnsManaged button').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      premigDnsManaged = btn.dataset.value;
      $('premig_remoteWrap').style.display = premigDnsManaged === 'remote' ? 'block' : 'none';
      generate();
    });
  });

  // Conditional listeners
  $('premig_customFindingToggle').addEventListener('change', ()=>{
    $('premig_customFindingWrap').style.display =$('premig_customFindingToggle').checked ? 'block' : 'none';
    $('premig_countsWrap').style.display = $('premig_customFindingToggle').checked ? 'none' : ($('premig_includeCounts').checked ? 'block' : 'none');
    generate();
  });
  $('premig_includeCounts').addEventListener('change', ()=>{
    $('premig_countsWrap').style.display =$('premig_includeCounts').checked ? 'block' : 'none';
    generate();
  });
  $('premig_titanEnable').addEventListener('change', ()=>{
    $('premig_titanWrap').style.display =$('premig_titanEnable').checked ? 'block' : 'none';
    generate();
  });
  $('premig_titanMultiple').addEventListener('change', ()=>{
    $('premig_titanEmailWrap').style.display =$('premig_titanMultiple').checked ? 'none' : 'block';
    generate();
  });
  $('premig_recordsNow').addEventListener('change', ()=>{
    $('premig_recordsNowWrap').style.display =$('premig_recordsNow').checked ? 'block' : 'none';
    generate();
  });
  $('premig_titanPreset').addEventListener('click', ()=>{$('premig_records').value = 'MX Records:\n\nHost/Name: @ | Type: MX | Priority: 10 | Value: mx1.titan.email\nHost/Name: @ | Type: MX | Priority: 20 | Value: mx2.titan.email\n\nSPF Record:\n\nHost/Name: @ | Type: TXT | Value: v=spf1 include:spf.titan.email ~all';
    generate();
  });
  $('premig_includeSignoff').addEventListener('change', ()=>{
    $('premig_signoffWrap').style.display =$('premig_includeSignoff').checked ? 'block' : 'none';
    generate();
  });
  $('premig_scheduled').addEventListener('change', ()=>{
    $('premig_scheduleFields').style.display =$('premig_scheduled').checked ? 'block' : 'none';
    generate();
  });

  // Bind inputs
  document.querySelectorAll('input[type=text], input[type=date], select, textarea, input[type=checkbox]').forEach(el=>{
    el.addEventListener('input', generate);
    el.addEventListener('change', generate);
  });

  // Copy / Reset
  $('copyBtn').addEventListener('click', async ()=>{
    const text = $('output').textContent;
    try{ await navigator.clipboard.writeText(text); }catch(e){}
    const btn = $('copyBtn');
    const original = btn.textContent;
    btn.textContent = 'Copied';
    btn.classList.add('copied');
    setTimeout(()=>{ btn.textContent = original; btn.classList.remove('copied'); }, 1600);
  });

  $('resetBtn').addEventListener('click', ()=>{
    document.querySelectorAll('input[type=text]').forEach(el=>el.value='');
    document.querySelectorAll('textarea').forEach(el=>el.value='');
    document.querySelectorAll('input[type=date]').forEach(el=>el.value='');
